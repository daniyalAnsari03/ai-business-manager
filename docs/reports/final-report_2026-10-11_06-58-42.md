## 5. Post-apply verification SQL (read-only)

Run only after the owner applies the migration, against the target DB:

```sql
-- 1. New tables exist
select table_name from information_schema.tables
where table_schema='public' and table_name in ('suppliers','supplier_bills','payments')
order by table_name;

-- 2. New functions are SECURITY DEFINER and have a locked search_path
select p.proname, p.prosecdef, p.proconfig
from pg_proc p join pg_namespace n on n.oid = p.pronamespace
where n.nspname='public'
  and p.proname in ('record_payment','reverse_payment','create_supplier_bill','update_supplier_bill',
                    'delete_supplier_bill','record_payment_for_business','reverse_payment_for_business',
                    'get_order_payment_summary','list_order_payment_summaries')
order by p.proname;

-- 3. Execute privileges per role (wrappers: authenticated; *_for_business: service_role only)
select p.proname,
       has_function_privilege('anon',          p.oid,'EXECUTE') as anon_can,
       has_function_privilege('authenticated', p.oid,'EXECUTE') as auth_can,
       has_function_privilege('service_role',  p.oid,'EXECUTE') as service_can
from pg_proc p join pg_namespace n on n.oid = p.pronamespace
where n.nspname='public'
  and p.proname in ('record_payment','reverse_payment','record_payment_for_business','reverse_payment_for_business')
order by p.proname;

-- 4. Unique indexes (idempotency keys, supplier name per business)
select i.relname as index_name, t.relname as table_name, x.indisunique, pg_get_indexdef(x.indexrelid)
from pg_index x
join pg_class i on i.oid = x.indexrelid
join pg_class t on t.oid = x.indrelid
join pg_namespace n on n.oid = t.relnamespace
where n.nspname='public' and t.relname in ('payments','suppliers','supplier_bills')
order by t.relname, i.relname;

-- 5. RLS policies on the new tables
select tablename, policyname, cmd, roles, qual, with_check
from pg_policies
where schemaname='public' and tablename in ('suppliers','supplier_bills','payments')
order by tablename, policyname;

-- 6. Triggers on payments and orders
select event_object_table, trigger_name, event_manipulation, action_timing
from information_schema.triggers
where trigger_schema='public' and event_object_table in ('payments','orders')
order by event_object_table, trigger_name;

-- 7. opening_paid backfill: pre-existing orders carry it; nothing created after does
select count(*) filter (where opening_paid is not null and opening_paid <> 0) as orders_with_opening_paid,
       count(*) as total_orders
from public.orders;
```
